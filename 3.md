## SEGMENT 1: useEffect Fundamentals 


**Live code — a live clock, side-effect-only, no fetching yet:**

```jsx
import { useState, useEffect } from 'react';

function Clock() {
  const [time, setTime] = useState(new Date());

  useEffect(() => {
    const timer = setInterval(() => {
      setTime(new Date());
    }, 1000);

    // cleanup function — runs when component unmounts
    return () => clearInterval(timer);
  }, []); // empty array = run once, on mount

  return <p>Current time: {time.toLocaleTimeString()}</p>;
}

export default Clock;
```



```jsx
// ❌ DANGER — no dependency array, this re-fetches on EVERY render,
// which if it updates state, triggers another render, which fetches again...
useEffect(() => {
  fetchSomething();
}); // no array at all

// ✅ SAFE — runs once
useEffect(() => {
  fetchSomething();
}, []);
```

    <ul>
      {users.map((user) => (

    </ul>
  );
}

export default UserList;

## SEGMENT 3: Loading & Error States

```jsx
import { useState, useEffect } from 'react';
import axios from 'axios';

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    axios.get('https://jsonplaceholder.typicode.com/users')
      .then((response) => {
        setUsers(response.data);
      })
      .catch((err) => {
        setError(err.message);
      })
      .finally(() => {
        setLoading(false);
      });
  }, []);

  if (loading) return <p>Loading users...</p>;
  if (error) return <p>Something went wrong: {error}</p>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name} — {user.email}</li>
      ))}
    </ul>
  );
}

export default UserList;
```

**common errors:**

| Error | Cause | Fix |
|---|---|---|
| Infinite re-fetch / frozen tab | Missing or wrong `useEffect` dependency array | Confirm `[]` for run-once |
| "Cannot read properties of undefined" | Rendering `.map()` before data arrives | Add `if (loading) return ...` before the main return |
| CORS error in console | Fetching own Spring Boot backend without CORS config | Stick to the public test API today; CORS setup is a separate backend-side fix |
| Blank white page after adding Router | `BrowserRouter` not wrapping `App` in `main.jsx` | Check `main.jsx` wraps `<App />` in `<BrowserRouter>` |
| Full page reload/flicker on nav | Used `<a href>` instead of `<Link to>` | Swap to `Link` |
| Retry button doesn't work | Fetch logic only exists inside `useEffect`, no reusable function to call again | Pull the fetch logic into a named function, call it both in `useEffect` and from the retry button's `onClick` |

**Retry button hint:**

```jsx
function fetchUsers() {
  setLoading(true);
  setError(null);
  axios.get('https://jsonplaceholder.typicode.com/users')
    .then((response) => setUsers(response.data))
    .catch((err) => setError(err.message))
    .finally(() => setLoading(false));
}

useEffect(() => {
  fetchUsers();
}, []);

// in JSX:
{error && <button onClick={fetchUsers}>Retry</button>}
```
