## SEGMENT 1: useState

**Live code — the classic counter:**

```jsx
import { useState } from 'react';

function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>Add</button>
    </div>
  );
}

export default Counter;
```



```
❌ count = count + 1        // React doesn't know this happened
✅ setCount(count + 1)      // React re-renders because you told it to
```

**Second example — state with an object (relevant for their Team Roster later):**

```jsx
function LikeButton() {
  const [liked, setLiked] = useState(false);

  return (
    <button onClick={() => setLiked(!liked)}>
      {liked ? '❤️ Liked' : '🤍 Like'}
    </button>
  );
}
```

## SEGMENT 2: Event Handlers 

**Live code — text input tied to state (controlled input):**

```jsx
import { useState } from 'react';

function NameForm() {
  const [name, setName] = useState('');

  function handleChange(event) {
    setName(event.target.value);
  }

  return (
    <div>
      <input type="text" value={name} onChange={handleChange} />
      <p>Hello, {name || 'stranger'}!</p>
    </div>
  );
}

export default NameForm;
```

**Common mistake to flag on the board:**

```jsx
// ❌ WRONG — calling the function immediately instead of passing a reference
<button onClick={handleClick()}>Click</button>

// ✅ RIGHT — pass the function, don't call it
<button onClick={handleClick}>Click</button>

// ✅ RIGHT — arrow function if you need to pass arguments
<button onClick={() => handleClick(id)}>Click</button>
```

## SEGMENT 3: Conditional Rendering 

**Pattern 1 — ternary:**

```jsx
function StatusBadge({ isOnline }) {
  return (
    <span>
      {isOnline ? '🟢 Online' : '⚫ Offline'}
    </span>
  );
}
```

**Pattern 2 — `&&` for show/nothing:**

```jsx
function Notification({ count }) {
  return (
    <div>
      <h3>Inbox</h3>
      {count > 0 && <p>You have {count} new messages</p>}
    </div>
  );
}
```


```jsx
// ❌ BUG — if count is 0, this renders the literal number 0 on screen
{count && <p>You have {count} messages</p>}

// ✅ FIX — force a real boolean
{count > 0 && <p>You have {count} messages</p>}
```


**toggle visibility with state:**

```jsx
import { useState } from 'react';

function Accordion() {
  const [open, setOpen] = useState(false);

  return (
    <div>
      <button onClick={() => setOpen(!open)}>
        {open ? 'Hide details' : 'Show details'}
      </button>
      {open && <p>Here are the hidden details you just revealed.</p>}
    </div>
  );
}
```

## SEGMENT 4: Lists with `.map()` and `key`

**Live code:**

```jsx
const people = [
  { id: 1, name: 'Kiruthik Kumar', role: 'Trainer' },
  { id: 2, name: 'Anish', role: 'Developer' },
  { id: 3, name: 'Priya', role: 'Designer' },
];

function App() {
  return (
    <div>
      {people.map((person) => (
        <ProfileInfo key={person.id} name={person.name} role={person.role} />
      ))}
    </div>
  );
}
```


```jsx
// ⚠️ Works but fragile — avoid if list order can change
{people.map((person, index) => (
  <ProfileInfo key={index} ... />
))}

// ✅ Better — stable, unique, tied to the actual data
{people.map((person) => (
  <ProfileInfo key={person.id} ... />
))}
```

 convert the hardcoded 4 `TeamMember` tags into a `.map()` over an array of person objects.
